```c
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
```

```c
// 主循环中
while (1)
```

```c
#include <stdio.h>

void parse_command(char* cmd)
{
    if (strstr(cmd, "led on"))
    {
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_RESET);
        printf("LED ON\r\n");
    }
    else if (strstr(cmd, "led off"))
    {
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_SET);
        printf("LED OFF\r\n");
    }
    else if (strstr(cmd, "led blink"))
    {
        printf("LED BLINK MODE\r\n");
        // 实现闪烁逻辑
    }
    else
    {
        printf("Unknown command: %s\r\n", cmd);
    }
}

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_USART1_UART_Init();

    printf("=== CMD Console ===\r\n");
    printf("Commands: led on/off/blink\r\n");
    printf(">\r\n");

    uint8_t rx_byte;
    uint8_t rx_index = 0;

    while (1)
    {
        if (HAL_UART_Receive(&huart1, &rx_byte, 1, 1000) == HAL_OK)
        {
            HAL_UART_Transmit(&huart1, &rx_byte, 1, 0);  // 回显
            
            if (rx_byte == '\r' || rx_byte == '\n')
            {
                rx_buffer[rx_index] = '\0';
                parse_command(rx_buffer);
                rx_index = 0;
                printf(">\r\n");
            }
            else if (rx_index < 63)
            {
                rx_buffer[rx_index++] = rx_byte;
            }
        }
    }
}
```

### 5.1 需求说明

### 5.2 硬件清单
| 材料 | 数量 | 备注 |
|------|------|------|
| STM32F103C8T6最小系统板 | 1块 | |
| HC-SR04超声波模块 | 1个 | |

### 5.3 硬件连接
```
HC-SR04    STM32F103
VCC    →   5V (或3.3V)
GND    →   GND
Trig   →   PA5
Echo   →   PA6
```

### 5.4 核心代码

```c
#define TRIG_PIN GPIO_PIN_5
#define ECHO_PIN GPIO_PIN_6

// 发送触发信号
void HCSR04_Trigger(void)
{
    HAL_GPIO_WritePin(GPIOA, TRIG_PIN, GPIO_PIN_SET);
    HAL_Delay_us(10);  // 10us高电平
    HAL_GPIO_WritePin(GPIOA, TRIG_PIN, GPIO_PIN_RESET);
}

// 测量回波高电平时间(单位: us)
uint32_t HCSR04_Measure(void)
{
    uint32_t start_ticks, end_ticks;
    
    // 等待Echo变为高电平
    while (HAL_GPIO_ReadPin(GPIOA, ECHO_PIN) == GPIO_PIN_RESET);
    start_ticks = HAL_GetTick() * 1000 + SysTick->VAL / 72;
    
    // 等待Echo变为低电平
    while (HAL_GPIO_ReadPin(GPIOA, ECHO_PIN) == GPIO_PIN_SET);
    end_ticks = HAL_GetTick() * 1000 + SysTick->VAL / 72;
    
    // 计算时间差 (us)
    return (end_ticks - start_ticks);
}

// 计算距离: 距离 = 时间(us) / 58 (cm)
float HCSR04_GetDistance(void)
{
    HCSR04_Trigger();
    HAL_Delay(60);  // 等待测量完成
    uint32_t echo_time = HCSR04_Measure();
    return echo_time / 58.0f;
}
```

## ⭐⭐⭐ 项目7: FreeRTOS多任务协同

### 7.1 需求说明
创建3个任务：
- 任务1: LED每秒闪烁一次
- 任务2: 按键检测，按下时发送消息到队列
- 任务3: 从队列接收消息，通过串口打印

### 7.2 核心代码

```c
// 任务句柄
TaskHandle_t xLedTaskHandle;
TaskHandle_t xKeyTaskHandle;
TaskHandle_t xDisplayTaskHandle;

// 消息队列
QueueHandle_t xKeyQueue;

// LED任务
void vLedTask(void *pvParameters)
{
    while (1)
    {
        HAL_GPIO_TogglePin(GPIOC, GPIO_PIN_13);
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

// 按键任务
void vKeyTask(void *pvParameters)
{
    uint8_t key_event = 0;
    while (1)
    {
        if (HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0) == GPIO_PIN_RESET)
        {
            vTaskDelay(pdMS_TO_TICKS(20));  // 消抖
            if (HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0) == GPIO_PIN_RESET)
            {
                key_event++;
                xQueueSend(xKeyQueue, &key_event, 0);
                while (HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0) == GPIO_PIN_RESET);
            }
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

// 显示任务
void vDisplayTask(void *pvParameters)
{
    uint8_t count = 0;
    while (1)
    {
        if (xQueueReceive(xKeyQueue, &count, portMAX_DELAY) == pdTRUE)
        {
            printf("Key pressed! Count: %d\r\n", count);
        }
    }
}

// 主函数中创建任务
void MX_FREERTOS_Init(void)
{
    xKeyQueue = xQueueCreate(10, sizeof(uint8_t));
    
    xTaskCreate(vLedTask, "LED", 128, NULL, 1, &xLedTaskHandle);
    xTaskCreate(vKeyTask, "KEY", 128, NULL, 2, &xKeyTaskHandle);
    xTaskCreate(vDisplayTask, "DISP", 256, NULL, 1, &xDisplayTaskHandle);
}
```

```c
    mqtt_publish(temp, humi, light);
    
    // 订阅控制命令
    client.subscribe("home/control");
    client.loop();
    
    delay(5000);  // 5秒上报一次
}
```
