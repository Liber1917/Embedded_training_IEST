// 实现 RAII 锁守卫
template<typename Mutex>
class LockGuard {
public:
    explicit LockGuard(Mutex& m) : mutex_(&m) { mutex_->lock(); }
    ~LockGuard() { mutex_->unlock(); }
    // 禁止拷贝和移动
private:
    Mutex* mutex_;
};
